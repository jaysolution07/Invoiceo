const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheetComptes = ss.getSheetByName("Comptes"); function doGet(e) { const params = (e && e.parameter) ? e.parameter : {}; const callback = params.callback || ""; const response = handleRequest(params); const json = JSON.stringify(response); // JSONP classique (Android, Chrome) if (callback && callback !== "_fetch_") { return ContentService .createTextOutput(`${callback}(${json})`) .setMimeType(ContentService.MimeType.JAVASCRIPT); } // JSON pur avec headers CORS (Safari iPhone) return ContentService .createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); } function doOptions(e) { return ContentService .createTextOutput("") .setMimeType(ContentService.MimeType.TEXT); } function handleRequest(params) { try { if (params.login == "1") { const email = (params.email || "").toLowerCase().trim(); const mdp = (params.mdp || "").trim(); if (!email || !mdp) return { success: false, error: "Paramètres manquants" }; const lastRow = sheetComptes.getLastRow(); if (lastRow < 2) return { success: false, error: "Aucun compte" }; const rows = sheetComptes.getRange("A2:E" + lastRow).getValues(); const found = rows.find(row => row[1].toString().toLowerCase().trim() === email && row[2].toString().trim() === mdp ); if (found) { const couleur = String(found[4] || "1").replace(".0", "").trim(); return { success: true, name: String(found[0] || email), webapp_url: String(found[3]), couleur: couleur || "1" }; } return { success: false, error: "Identifiants incorrects" }; } return { error: "Paramètre inconnu" }; } catch(err) { return { success: false, error: err.toString() }; } } function debugLogin() { const rows = sheetComptes.getRange("A2:E" + sheetComptes.getLastRow()).getValues(); rows.forEach((row, i) => { Logger.log(`Row ${i}: email="${row[1]}" mdp="${row[2]}"`); }); }
